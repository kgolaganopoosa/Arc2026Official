Supabase Cloud Connection

Local Cache Mode

Connect your Supabase project to sync customers, Black Card rewards, couriers, notifications, and admin settings in real time. Hidden from public visitors.

One-Click Supabase SQL Schema Setup

Run this SQL in your Supabase SQL Editor to create all required tables and seed the admin security password.

-- Shop MODISHA Supabase Schema
CREATE TABLE IF NOT EXISTS app_settings (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL
);

INSERT INTO app_settings (key, value)
VALUES ('admin_password', '#PetalChain_25')
ON CONFLICT (key) DO NOTHING;

CREATE TABLE IF NOT EXISTS customers (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  black_card_id TEXT UNIQUE NOT NULL,
  pin TEXT NOT NULL,
  whatsapp TEXT UNIQUE NOT NULL,
  rewards NUMERIC(10,2) DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS couriers (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  whatsapp TEXT UNIQUE NOT NULL,
  email TEXT NOT NULL,
  pin TEXT NOT NULL,
  bicycle_img TEXT,
  selfie_img TEXT,
  total_earnings NUMERIC(10,2) DEFAULT 0,
  trips INTEGER DEFAULT 0,
  gold_rewards NUMERIC(10,2) DEFAULT 0,
  suspended BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS notifications (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  recipient_type TEXT NOT NULL, -- 'customer' or 'courier'
  recipient_id TEXT NOT NULL,   -- black_card_id or courier whatsapp
  message TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS orders (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  customer_name TEXT NOT NULL,
  black_card_id TEXT NOT NULL,
  whatsapp TEXT NOT NULL,
  items TEXT NOT NULL,
  cash_amount NUMERIC(10,2) DEFAULT 0,
  payment_method TEXT NOT NULL,
  rewards_spent NUMERIC(10,2) DEFAULT 0,
  discount_code TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
